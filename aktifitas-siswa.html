Ingat Kembali

Struktur data abstrak (ADT) adalah tipe data buatan sendiri yang terdiri dari:

  • data (representasi): apa saja yang disimpan, dan
  • operasi: apa saja yang dapat dilakukan terhadap data tersebut.

Dalam C++, data ADT dituliskan dengan struct, dan operasinya dituliskan sebagai fungsi atau prosedur.

#include <iostream>
#include <string>
using namespace std;

struct Siswa {          // data ADT
    string nama;
    int nilai;
};

void tampilkan(Siswa s) {   // operasi ADT
    cout << s.nama << " : " << s.nilai << endl;
}

int main() {
    Siswa s1;
    s1.nama = "Ani";        // mengakses anggota dengan tanda titik
    s1.nilai = 80;
    tampilkan(s1);
    return 0;
}
  • Siswa = nama tipe baru yang dibuat dengan struct
  • nama dan nilai = anggota (field) dari struct
  • s1.nama = cara mengakses anggota nama milik variabel s1

#include <iostream>
using namespace std;

struct Pecahan {
    int pembilang;
    int penyebut;
};

Pecahan kali(Pecahan p, Pecahan q) {
    Pecahan hasil;
    hasil.pembilang = p.pembilang * q.pembilang;
    hasil.penyebut  = p.penyebut * q.penyebut;
    return hasil;
}

int main() {
    Pecahan a = {1, 2};
    Pecahan b = {3, 4};
    Pecahan c = kali(a, b);

}
